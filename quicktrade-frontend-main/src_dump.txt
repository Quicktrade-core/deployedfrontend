import OrderPage from "./pages/orderPage"
import ProfilePage from "./pages/profilePage"
import SecurityPage from "./pages/securityPage"

//import VideoPage from "./pages/videoPage"
import Products from "./components/products"
import Reels from "./components/reels"
import Stores from "./components/stores"
import Loading from "./components/loading"
import ContextProvider from "./context/contextProvider"

function App() {
  const [load, setLoad] = useState(true)
  const [open, setOpen] = useState(false)
  const [cart, setCart] = useState(false)

  useEffect(() => {
    const timer = setTimeout(() => {
      setLoad(false)
    }, 3000)

    return () => clearTimeout(timer)
  }, [])

  const router = createBrowserRouter([
    {
      path: `/`,
      element: <RootLayout />,
      children: [
        { index: true, element: <HomePage /> },
        { path: `/account`, element: <AccountPage /> }
      ]
    },
    {
      path: `/seller`,
      element: <SellerPage />,
      children: [
        { index: true, element: <Products /> },
        { path: `reels`, element: <Reels /> },
        { path: `stores`, element: <Stores /> }
      ]
    },
    { path: `/product/:id`, element: <ProductPage /> },
    { path: `/cart`, element: <CartPage /> },
    { path: `/checkout`, element: <CheckoutPage /> },
    { path: `/videos`, element: <VideosPage /> },
    // { path: `/video/:id`, element: <VideoPage /> },
    { path: `/profile`, element: <ProfilePage /> },
    { path: `/orders`, element: <OrdersPage /> },
    { path: `/order/:id`, element: <OrderPage /> },
    { path: `/notifications`, element: <NotificationsPage /> },
    { path: `/security`, element: <SecurityPage /> },
    { path: `/settings`, element: <SettingsPage /> },
    { path: `/login`, element: <LoginPage /> },
    { path: `*`, element: <NotfoundPage /> }
  ])

  if (load) return <Loading />

  return (
    <ContextProvider>
      <RouterProvider router={router} />
    </ContextProvider>
  )
}

export default App

import { useContext, useState } from "react"
import { cartContext } from "../context/cartContext"
import { BiMinus, BiPlus, BiX } from "react-icons/bi"
import { Link } from "react-router"

const CartItem = ({ item }) => {
  const [quantity, setQuantity] = useState(1)
  const { cartItems, setCartItems } = useContext(cartContext)

  const removeFromCart = (id) => {
    setCartItems(previous => previous.filter(item => item._id !== id))
  }

  const decrease = (id) => {
    quantity > 1 ? setQuantity(previous => previous - 1) : removeFromCart(id)
  }

  const increase = () => {
    setQuantity(previous => previous + 1)
  }

  return (
    <div className="flex gap-5">
      <Link to={`/product/${item._id}`} className="w-1/3 h-20">
        <img src={item.image} alt={item.name} className="w-full h-full object-cover rounded-xl" />
      </Link>
      <div className="w-2/3 flex flex-col justify-between">
        <div className="flex items-center gap-5">
          <p className="leading-tight">{item.name}</p>
          <button className="cursor-pointer">
            <BiX onClick={() => { removeFromCart(item._id) }} className="text-2xl" />
          </button>
        </div>
        <div className="flex justify-between items-center">
          <p className="w-1/2 font-semibold">${item.price}</p>
          <div className="flex justify-between items-center border-2 border-neutral-200 rounded-full w-1/2 px-2">
            <button className="cursor-pointer">
              <BiMinus onClick={() => { decrease(item._id) }} />
            </button>
            {quantity}
            <button className="cursor-pointer">
              <BiPlus onClick={increase} />
            </button>
          </div>
        </div>
      </div>
    </div>
  )
}

export default CartItem

const VideoPlayer = ({ src, videoData, muted, setMuted }) => {
  const videoRef = useRef(null)
  const [like, setLike] = useState(false)
  const [following, setFollowing] = useState(false)
  const style = `bg-none border-none text-white flex flex-col items-center cursor-pointer`

  const toggleMute = () => {
    setMuted(previous => !previous)
  }

  useEffect(() => {
    const video = videoRef.current;
    const observer = new IntersectionObserver(
      ([entry]) => {
        if (entry.isIntersecting) {
          video.play().catch(() => {});
        } else {
          video.pause();
        }
      },
      {
        threshold: 0.7,
      }
    );

    observer.observe(video);

    return () => {
      observer.disconnect();

  return (
